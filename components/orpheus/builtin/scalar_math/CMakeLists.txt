cmake_minimum_required(VERSION 3.16)
project(orpheus_builtin_scalar_math C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_LIST_DIR}/generated/component_sources.cmake")
add_library(orpheus_builtin_scalar_math SHARED ${ORPHEUS_COMPONENT_SOURCES})
target_include_directories(orpheus_builtin_scalar_math PUBLIC ${ORPHEUS_COMPONENT_INCLUDE_DIRS})
if(TARGET orpheus_abi)
    target_link_libraries(orpheus_builtin_scalar_math PUBLIC orpheus_abi)
else()
    set(ORPHEUS_ABI_INCLUDE_DIR "" CACHE PATH "orpheus_abi.h 所在目录")
    if(NOT EXISTS "${ORPHEUS_ABI_INCLUDE_DIR}/orpheus_abi.h")
        message(FATAL_ERROR "请设置 ORPHEUS_ABI_INCLUDE_DIR")
    endif()
    target_include_directories(orpheus_builtin_scalar_math PUBLIC "${ORPHEUS_ABI_INCLUDE_DIR}")
endif()
# 空生成器表达式避免多配置构建器自动追加 Debug/Release 子目录。
set_target_properties(orpheus_builtin_scalar_math PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/components$<0:>"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/components$<0:>"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/components$<0:>")
target_compile_definitions(orpheus_builtin_scalar_math PRIVATE ORPHEUS_BUILDING_COMPONENT)
if(MSVC)
    target_compile_options(orpheus_builtin_scalar_math PRIVATE /utf-8)
endif()
